循环结构对比

CC — while(true) 生成器

async function* queryLoop(params) {
  let state = { messages, toolUseContext, turnCount, ... }

  while (true) {                         // 无限循环
    // ① 流式调用 LLM API
    // ② StreamingToolExecutor 边收边执行工具
    // ③ 收集所有工具结果

    if (!has_tool_use)   return 'completed'     // 正常：模型说完
    if (user_aborted)    return 'aborted_tools'  // 异常：用户中断
    if (hook_stopped)    return 'hook_stopped'   // 异常：Hook 阻止
    if (max_turns)       return 'max_turns'      // 异常：上限
    if (overflow)        return 'prompt_too_long' // 异常：溢出

    // ④ 有工具结果 → 更新 state → continue
    state = { messages: [...prev, ...toolResults], turnCount++ }
  }
}

CX — loop + sampling_request

fn run_turn() {
    loop {
        // ① 从 input_queue 获取输入
        // ② 运行 hooks → 若阻止则 break
        // ③ run_sampling_request() → 流式 LLM + 执行工具
        let needs_follow_up = ...;         // 工具是否需要后续

        // ④ Token 预算检查
        let should_roll_over = needs_follow_up
            && (new_context_window || token_limit_reached);

        if should_roll_over { /* 新上下文窗口 */ }
        else if needs_follow_up { continue }  // 继续循环
        else { break }                        // 正常退出
    }
}

fn try_run_sampling_request() {
    loop {
        let event = stream.next().await;
        match event {
            OutputItemDone(item) => handle_tool_call(item),
            // ... 有工具调用 → needs_follow_up = true
        }
    }
}

OX — Effect 双层 while

const run = Effect.fn(function* (input) {
  while (shouldRun) {                      // 外层：队列处理
    let step = 1
    while (needsContinuation) {            // 内层：agent 循环
      const result = yield* runTurn(sessionID, promotion, step)
      needsContinuation = result.needsContinuation  // 工具调用→继续
      step++
      if (!needsContinuation)
        needsContinuation = yield* hasPending("steer")
    }
    shouldRun = yield* hasPending("queue")
  }
})

共同模式：三个 agent 都采用"while 循环 + 流式 LLM + 工具并行执行"的基本架构。区别在于：CC 用 async generator 可被外部消费；CX 是传统 loop + tokio 异步；OX 用 Effect-TS 函数式风格。

循环伪代码通用模式

所有三个 agent 的核心循环可抽象为: while (true) { response = callLLM(messages) // 流式调用 toolCalls = extractToolCalls(response) // 提取工具调用 if (toolCalls.length === 0) return // 正常退出 results = executeTools(toolCalls) // 并行/串行执行 messages.append(assistantMsg, results) // 拼入历史 // → 继续循环 } 异常退出: 中断 / Hook阻止 / 上限 / 溢出 / 错误
